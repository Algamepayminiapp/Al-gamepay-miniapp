var gameGrid = document.getElementById("gameGrid");
var homePage = document.getElementById("homePage");
var detailPage = document.getElementById("detailPage");
var searchInput = document.getElementById("searchInput");
var productName = document.getElementById("productName");
var productIcon = document.getElementById("productIcon");
var packageList = document.getElementById("packageList");
var totalPrice = document.getElementById("totalPrice");
var buyButton = document.getElementById("buyButton");

var selectedGame = null;
var selectedPackage = null;

function showGames(list) {
  gameGrid.innerHTML = "";

  list.forEach(function(game) {
    var button = document.createElement("button");
    button.className = "game";

    var region = "";

    if (game.regions > 0) {
      region =
        '<div class="region">REGIONS ' +
        game.regions +
        '</div>';
    }

    button.innerHTML =
      '<div class="cover">' +
        region +
        '<div class="cover-name">' +
          game.name +
        '</div>' +
      '</div>' +
      '<div class="game-title">' +
        game.name +
      '</div>';

    button.addEventListener("click", function() {
      openGame(game);
    });

    gameGrid.appendChild(button);
  });
}

function openGame(game) {
  selectedGame = game;
  selectedPackage = null;

  productName.textContent = game.name;
  productIcon.textContent = game.name.substring(0, 2).toUpperCase();

  packageList.innerHTML = "";
  totalPrice.textContent = "—";

  buyButton.disabled = true;
  buyButton.textContent = "Select a Package First";

  game.packages.forEach(function(item) {
    var packageName = item[0];
    var packagePrice = item[1];

    var button = document.createElement("button");
    button.className = "package";

    button.innerHTML =
      "<strong>" +
      packageName +
      "</strong>" +
      '<span class="price">' +
      packagePrice.toLocaleString() +
      " MMK</span>";

    button.addEventListener("click", function() {
      var allPackages =
        document.querySelectorAll(".package");

      allPackages.forEach(function(btn) {
        btn.classList.remove("active");
      });

      button.classList.add("active");

      selectedPackage = {
        name: packageName,
        price: packagePrice
      };

      totalPrice.textContent =
        packagePrice.toLocaleString() +
        " MMK";

      buyButton.disabled = false;
      buyButton.textContent = "Buy Now";
    });

    packageList.appendChild(button);
  });

  homePage.style.display = "none";
  detailPage.style.display = "block";

  window.scrollTo(0, 0);
}
